cmake_minimum_required(VERSION 3.22)

if (NOT DEFINED BLITZMAX_ESP32_ARENA_SIZE)
    message(FATAL_ERROR "BLITZMAX_ESP32_ARENA_SIZE must be supplied by bmk")
elseif (BLITZMAX_ESP32_ARENA_SIZE LESS_EQUAL 0)
    message(FATAL_ERROR "The BlitzMax ESP32 managed arena must be larger than zero")
endif()
math(EXPR BLITZMAX_ESP32_ARENA_REMAINDER "${BLITZMAX_ESP32_ARENA_SIZE} % 16")
if (NOT BLITZMAX_ESP32_ARENA_REMAINDER EQUAL 0)
    message(FATAL_ERROR "The BlitzMax ESP32 managed arena must be a multiple of 16 bytes")
endif()

list(FIND BLITZMAX_ESP32_MODULE_INITIALIZERS "__bb_embedded_network_mqtt_mqtt" BLITZMAX_ESP32_PROJECT_MQTT_INDEX)
if (NOT BLITZMAX_ESP32_PROJECT_MQTT_INDEX EQUAL -1)
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/optional_components")
endif()
if ("$ENV{IDF_COMPONENT_MANAGER}" STREQUAL "0" AND
        EXISTS "${CMAKE_CURRENT_LIST_DIR}/managed_components/joltwallet__littlefs")
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/managed_components")
endif()
list(FIND BLITZMAX_ESP32_MODULE_INITIALIZERS "__bb_embedded_network_thread_thread" BLITZMAX_ESP32_PROJECT_THREAD_INDEX)
if (NOT BLITZMAX_ESP32_PROJECT_THREAD_INDEX EQUAL -1)
    if (NOT IDF_TARGET STREQUAL "esp32h2" AND NOT IDF_TARGET STREQUAL "esp32c6" AND
            NOT IDF_TARGET STREQUAL "esp32s31")
        message(FATAL_ERROR "Embedded.Network.Thread requires an ESP32-H2, ESP32-C6, or ESP32-S31 board profile")
    endif()
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/thread.defaults")
    if (DEFINED ENV{BMX_ESP32_EXTRA_SDKCONFIG} AND NOT "$ENV{BMX_ESP32_EXTRA_SDKCONFIG}" STREQUAL "")
        list(APPEND SDKCONFIG_DEFAULTS "$ENV{BMX_ESP32_EXTRA_SDKCONFIG}")
    endif()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
list(FIND BLITZMAX_ESP32_MODULE_INITIALIZERS "__bb_embedded_network_wifi_wifi" BLITZMAX_ESP32_PROJECT_WIFI_INDEX)
list(FIND BLITZMAX_ESP32_MODULE_INITIALIZERS "__bb_embedded_network_ble_ble" BLITZMAX_ESP32_PROJECT_BLE_INDEX)
list(FIND BLITZMAX_ESP32_MODULE_INITIALIZERS "__bb_embedded_network_ethernet_ethernet" BLITZMAX_ESP32_PROJECT_ETHERNET_INDEX)
list(FIND BLITZMAX_ESP32_MODULE_INITIALIZERS "__bb_pub_net_net" BLITZMAX_ESP32_PROJECT_NET_INDEX)
list(FIND BLITZMAX_ESP32_MODULE_INITIALIZERS "__bb_esp32_hardware_rmt_rmt" BLITZMAX_ESP32_PROJECT_RMT_INDEX)
if (NOT BLITZMAX_ESP32_PROJECT_WIFI_INDEX EQUAL -1 AND NOT BLITZMAX_ESP32_PROJECT_BLE_INDEX EQUAL -1)
    set(COMPONENTS main bt esp_event esp_netif esp_wifi nvs_flash)
    set(BLITZMAX_ESP32_WIFI_ENABLED TRUE CACHE BOOL "Build the BlitzMax ESP32 Wi-Fi adapter" FORCE)
    set(BLITZMAX_ESP32_BLE_ENABLED TRUE CACHE BOOL "Build the BlitzMax ESP32 BLE adapter" FORCE)
    set(BLITZMAX_ESP32_NET_ENABLED TRUE CACHE BOOL "Build BlitzMax socket support" FORCE)
elseif (NOT BLITZMAX_ESP32_PROJECT_WIFI_INDEX EQUAL -1)
    # Component discovery precedes the second evaluation of main/CMakeLists.
    # Seed the radio stack here so conditional main-component dependencies are
    # visible during ESP-IDF's minimal-build discovery pass.
    set(COMPONENTS main esp_event esp_netif esp_wifi nvs_flash)
    set(BLITZMAX_ESP32_WIFI_ENABLED TRUE CACHE BOOL "Build the BlitzMax ESP32 Wi-Fi adapter" FORCE)
    set(BLITZMAX_ESP32_BLE_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 BLE adapter" FORCE)
    set(BLITZMAX_ESP32_NET_ENABLED TRUE CACHE BOOL "Build BlitzMax socket support" FORCE)
elseif (NOT BLITZMAX_ESP32_PROJECT_BLE_INDEX EQUAL -1)
    set(COMPONENTS main bt nvs_flash)
    set(BLITZMAX_ESP32_WIFI_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 Wi-Fi adapter" FORCE)
    set(BLITZMAX_ESP32_BLE_ENABLED TRUE CACHE BOOL "Build the BlitzMax ESP32 BLE adapter" FORCE)
    set(BLITZMAX_ESP32_NET_ENABLED FALSE CACHE BOOL "Build BlitzMax socket support" FORCE)
elseif (NOT BLITZMAX_ESP32_PROJECT_NET_INDEX EQUAL -1)
    set(COMPONENTS main esp_netif lwip)
    set(BLITZMAX_ESP32_WIFI_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 Wi-Fi adapter" FORCE)
    set(BLITZMAX_ESP32_BLE_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 BLE adapter" FORCE)
    set(BLITZMAX_ESP32_NET_ENABLED TRUE CACHE BOOL "Build BlitzMax socket support" FORCE)
else()
    if (BLITZMAX_ESP32_PROJECT_RMT_INDEX EQUAL -1 AND BLITZMAX_ESP32_PROJECT_ETHERNET_INDEX EQUAL -1)
        idf_build_set_property(MINIMAL_BUILD ON)
    endif()
    set(BLITZMAX_ESP32_WIFI_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 Wi-Fi adapter" FORCE)
    set(BLITZMAX_ESP32_BLE_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 BLE adapter" FORCE)
    set(BLITZMAX_ESP32_NET_ENABLED FALSE CACHE BOOL "Build BlitzMax socket support" FORCE)
endif()
if (NOT BLITZMAX_ESP32_PROJECT_ETHERNET_INDEX EQUAL -1)
    if (DEFINED COMPONENTS)
        list(APPEND COMPONENTS esp_eth esp_event esp_netif lwip)
    else()
        set(COMPONENTS main esp_eth esp_event esp_netif lwip)
    endif()
    set(BLITZMAX_ESP32_ETHERNET_ENABLED TRUE CACHE BOOL "Build the BlitzMax ESP32 Ethernet adapter" FORCE)
else()
    set(BLITZMAX_ESP32_ETHERNET_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 Ethernet adapter" FORCE)
endif()
if (NOT BLITZMAX_ESP32_PROJECT_MQTT_INDEX EQUAL -1)
    if (DEFINED COMPONENTS)
        list(APPEND COMPONENTS bmx_mqtt_dependency espressif__mqtt esp_event esp_netif lwip)
    else()
        set(COMPONENTS main bmx_mqtt_dependency espressif__mqtt esp_event esp_netif lwip)
    endif()
endif()
if (NOT BLITZMAX_ESP32_PROJECT_RMT_INDEX EQUAL -1)
    # Component discovery runs before main/CMakeLists can add its conditional
    # dependency. Seed the driver here for applications that import RMT.
    if (DEFINED COMPONENTS)
        list(APPEND COMPONENTS esp_driver_rmt)
    else()
        set(COMPONENTS main esp_driver_rmt)
    endif()
endif()
if (NOT BLITZMAX_ESP32_PROJECT_THREAD_INDEX EQUAL -1)
    if (DEFINED COMPONENTS)
        list(APPEND COMPONENTS openthread esp_event esp_netif lwip nvs_flash)
    else()
        set(COMPONENTS main openthread esp_event esp_netif lwip nvs_flash)
    endif()
endif()
project(${BLITZMAX_OUTPUT_NAME})
