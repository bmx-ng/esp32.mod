cmake_minimum_required(VERSION 3.22)

if (NOT DEFINED BLITZMAX_ESP32_ARENA_SIZE)
    message(FATAL_ERROR "BLITZMAX_ESP32_ARENA_SIZE must be supplied by bmk")
elseif (BLITZMAX_ESP32_ARENA_SIZE LESS_EQUAL 0)
    message(FATAL_ERROR "The BlitzMax ESP32 managed arena must be larger than zero")
endif()
math(EXPR BLITZMAX_ESP32_ARENA_REMAINDER "${BLITZMAX_ESP32_ARENA_SIZE} % 16")
if (NOT BLITZMAX_ESP32_ARENA_REMAINDER EQUAL 0)
    message(FATAL_ERROR "The BlitzMax ESP32 managed arena must be a multiple of 16 bytes")
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
list(FIND BLITZMAX_ESP32_MODULE_INITIALIZERS "__bb_embedded_network_wifi_wifi" BLITZMAX_ESP32_PROJECT_WIFI_INDEX)
list(FIND BLITZMAX_ESP32_MODULE_INITIALIZERS "__bb_embedded_network_ble_ble" BLITZMAX_ESP32_PROJECT_BLE_INDEX)
list(FIND BLITZMAX_ESP32_MODULE_INITIALIZERS "__bb_pub_net_net" BLITZMAX_ESP32_PROJECT_NET_INDEX)
if (NOT BLITZMAX_ESP32_PROJECT_WIFI_INDEX EQUAL -1 AND NOT BLITZMAX_ESP32_PROJECT_BLE_INDEX EQUAL -1)
    set(COMPONENTS main bt esp_event esp_netif esp_wifi nvs_flash)
    set(BLITZMAX_ESP32_WIFI_ENABLED TRUE CACHE BOOL "Build the BlitzMax ESP32 Wi-Fi adapter" FORCE)
    set(BLITZMAX_ESP32_BLE_ENABLED TRUE CACHE BOOL "Build the BlitzMax ESP32 BLE adapter" FORCE)
    set(BLITZMAX_ESP32_NET_ENABLED TRUE CACHE BOOL "Build BlitzMax socket support" FORCE)
elseif (NOT BLITZMAX_ESP32_PROJECT_WIFI_INDEX EQUAL -1)
    # Component discovery precedes the second evaluation of main/CMakeLists.
    # Seed the radio stack here so conditional main-component dependencies are
    # visible during ESP-IDF's minimal-build discovery pass.
    set(COMPONENTS main esp_event esp_netif esp_wifi nvs_flash)
    set(BLITZMAX_ESP32_WIFI_ENABLED TRUE CACHE BOOL "Build the BlitzMax ESP32 Wi-Fi adapter" FORCE)
    set(BLITZMAX_ESP32_BLE_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 BLE adapter" FORCE)
    set(BLITZMAX_ESP32_NET_ENABLED TRUE CACHE BOOL "Build BlitzMax socket support" FORCE)
elseif (NOT BLITZMAX_ESP32_PROJECT_BLE_INDEX EQUAL -1)
    set(COMPONENTS main bt nvs_flash)
    set(BLITZMAX_ESP32_WIFI_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 Wi-Fi adapter" FORCE)
    set(BLITZMAX_ESP32_BLE_ENABLED TRUE CACHE BOOL "Build the BlitzMax ESP32 BLE adapter" FORCE)
    set(BLITZMAX_ESP32_NET_ENABLED FALSE CACHE BOOL "Build BlitzMax socket support" FORCE)
elseif (NOT BLITZMAX_ESP32_PROJECT_NET_INDEX EQUAL -1)
    set(COMPONENTS main esp_netif lwip)
    set(BLITZMAX_ESP32_WIFI_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 Wi-Fi adapter" FORCE)
    set(BLITZMAX_ESP32_BLE_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 BLE adapter" FORCE)
    set(BLITZMAX_ESP32_NET_ENABLED TRUE CACHE BOOL "Build BlitzMax socket support" FORCE)
else()
    idf_build_set_property(MINIMAL_BUILD ON)
    set(BLITZMAX_ESP32_WIFI_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 Wi-Fi adapter" FORCE)
    set(BLITZMAX_ESP32_BLE_ENABLED FALSE CACHE BOOL "Build the BlitzMax ESP32 BLE adapter" FORCE)
    set(BLITZMAX_ESP32_NET_ENABLED FALSE CACHE BOOL "Build BlitzMax socket support" FORCE)
endif()
project(${BLITZMAX_OUTPUT_NAME})
